cmake_minimum_required(VERSION 3.22)
project(NotbitDesktop VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets Test)
# Qt 6.8 references AGL, which Apple removed from the macOS 26 SDK.
# Qt Quick uses Metal; remove the unused legacy framework on those SDKs.
if(APPLE AND CMAKE_OSX_SYSROOT MATCHES "MacOSX26")
 get_target_property(qt_gui_links WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
 list(FILTER qt_gui_links EXCLUDE REGEX "AGL")
 set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${qt_gui_links}")
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
pkg_check_modules(SQLCIPHER REQUIRED IMPORTED_TARGET sqlcipher)
set(NTB ${CMAKE_CURRENT_SOURCE_DIR}/third_party/notbit)
add_library(notbit_codec STATIC
 ${NTB}/src/ntb-address.c ${NTB}/src/ntb-base58.c ${NTB}/src/ntb-buffer.c
 ${NTB}/src/ntb-util.c ${NTB}/src/ntb-error.c ${NTB}/src/ntb-ecc.c
 ${NTB}/src/ntb-proto.c ${NTB}/src/ntb-netaddress.c)
target_include_directories(notbit_codec PUBLIC ${NTB}/src ${NTB})
target_link_libraries(notbit_codec PUBLIC OpenSSL::Crypto)
target_compile_options(notbit_codec PRIVATE -Wno-deprecated-declarations -include sys/types.h)
add_library(bmcore src/storage.cpp src/protocol.cpp src/cache.cpp src/scanner.cpp)
target_include_directories(bmcore PUBLIC src)
target_link_libraries(bmcore PUBLIC notbit_codec Qt6::Core PkgConfig::SODIUM PkgConfig::SQLCIPHER OpenSSL::Crypto)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE bmcore)
enable_testing()
add_test(NAME core COMMAND core_tests)

target_compile_definitions(bmcore PUBLIC SQLITE_HAS_CODEC)
target_compile_options(bmcore PRIVATE -Wno-deprecated-declarations)

if(UNIX)
 file(GLOB NTB_ENGINE_SOURCES CONFIGURE_DEPENDS ${NTB}/src/*.c)
 list(FILTER NTB_ENGINE_SOURCES EXCLUDE REGEX "/(main|dump-store|ntb-address|ntb-base58|ntb-buffer|ntb-util|ntb-error|ntb-ecc|ntb-proto|ntb-netaddress)\\.c$")
 add_library(notbit_engine STATIC ${NTB_ENGINE_SOURCES})
 target_link_libraries(notbit_engine PUBLIC notbit_codec Threads::Threads)
 target_compile_options(notbit_engine PRIVATE -Wno-deprecated-declarations -include sys/types.h)
endif()
qt_add_executable(notbit-desktop MACOSX_BUNDLE src/main.cpp src/session.cpp src/session.h)
qt_add_resources(notbit-desktop "ui" PREFIX "/" FILES ui/Main.qml)
target_link_libraries(notbit-desktop PRIVATE bmcore Qt6::Quick Qt6::QuickControls2 Qt6::Widgets)
if(UNIX)
 target_link_libraries(notbit-desktop PRIVATE notbit_engine)
endif()
set_target_properties(notbit-desktop PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Notbit" MACOSX_BUNDLE_GUI_IDENTIFIER "org.notbit.desktop" MACOSX_BUNDLE_BUNDLE_VERSION "0.1.0")
install(TARGETS notbit-desktop BUNDLE DESTINATION . RUNTIME DESTINATION bin)
add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
target_link_libraries(lifecycle_tests PRIVATE bmcore)
add_test(NAME lifecycle COMMAND lifecycle_tests)
find_package(Python3 COMPONENTS Interpreter)
if(UNIX AND Python3_Interpreter_FOUND)
 add_test(NAME relay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relay_test.py $<TARGET_FILE:notbit-desktop>)
 set_tests_properties(relay PROPERTIES TIMEOUT 60)
endif()
add_test(NAME gui-smoke COMMAND notbit-desktop --offline --data-dir ${CMAKE_CURRENT_BINARY_DIR}/smoke-node --smoke-test)
set_tests_properties(gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(desktop_tests tests/desktop_tests.cpp src/session.cpp src/session.h)
qt_add_resources(desktop_tests "test-ui" PREFIX "/" FILES ui/Main.qml)
target_link_libraries(desktop_tests PRIVATE bmcore Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Test)
add_test(NAME desktop COMMAND desktop_tests)
set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
