cmake_minimum_required(VERSION 3.22)
project(ynotbit VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Widgets Network Test)
# Qt 6.8 references AGL, which Apple removed from the macOS 26 SDK.
# Qt Quick uses Metal; remove the unused legacy framework on those SDKs.
if(APPLE AND CMAKE_OSX_SYSROOT MATCHES "MacOSX26")
 get_target_property(qt_gui_links WrapOpenGL::WrapOpenGL INTERFACE_LINK_LIBRARIES)
 list(FILTER qt_gui_links EXCLUDE REGEX "AGL")
 set_property(TARGET WrapOpenGL::WrapOpenGL PROPERTY INTERFACE_LINK_LIBRARIES "${qt_gui_links}")
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
pkg_check_modules(SQLCIPHER REQUIRED IMPORTED_TARGET sqlcipher)
set(NTB ${CMAKE_CURRENT_SOURCE_DIR}/third_party/notbit)
add_library(notbit_codec STATIC
 ${NTB}/src/ntb-address.c ${NTB}/src/ntb-base58.c ${NTB}/src/ntb-buffer.c
 ${NTB}/src/ntb-util.c ${NTB}/src/ntb-error.c ${NTB}/src/ntb-ecc.c
 ${NTB}/src/ntb-proto.c ${NTB}/src/ntb-netaddress.c)
target_include_directories(notbit_codec PUBLIC ${NTB}/src ${NTB})
target_link_libraries(notbit_codec PUBLIC OpenSSL::Crypto)
target_compile_options(notbit_codec PRIVATE -Wno-deprecated-declarations -include sys/types.h)
add_library(bmcore src/storage.cpp src/mailbox_ops.cpp src/protocol.cpp src/protocol_wire.cpp src/pow.cpp src/delivery.cpp src/cache.cpp src/scanner.cpp)
target_include_directories(bmcore PUBLIC src)
target_link_libraries(bmcore PUBLIC Threads::Threads notbit_codec Qt6::Core PkgConfig::SODIUM PkgConfig::SQLCIPHER OpenSSL::Crypto)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE bmcore)
enable_testing()
add_test(NAME core COMMAND core_tests)

target_compile_definitions(bmcore PUBLIC SQLITE_HAS_CODEC)
target_compile_options(bmcore PRIVATE -Wno-deprecated-declarations)

if(UNIX)
 file(GLOB NTB_ENGINE_SOURCES CONFIGURE_DEPENDS ${NTB}/src/*.c)
 list(FILTER NTB_ENGINE_SOURCES EXCLUDE REGEX "/(main|dump-store|ntb-address|ntb-base58|ntb-buffer|ntb-util|ntb-error|ntb-ecc|ntb-proto|ntb-netaddress)\\.c$")
 add_library(notbit_engine STATIC ${NTB_ENGINE_SOURCES})
 target_link_libraries(notbit_engine PUBLIC notbit_codec Threads::Threads)
 target_compile_options(notbit_engine PRIVATE -Wno-deprecated-declarations -include sys/types.h)
endif()
qt_add_executable(ynotbit MACOSX_BUNDLE src/main.cpp src/session.cpp src/session.h
 src/portable_relay.cpp src/portable_relay.h src/appearance.cpp src/appearance.h
 src/markdown_editor.cpp src/markdown_editor.h)
qt_add_resources(ynotbit "ui" PREFIX "/" FILES ui/Main.qml)
target_link_libraries(ynotbit PRIVATE bmcore Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network)
if(UNIX)
 target_sources(ynotbit PRIVATE src/relay_bridge.cpp)
 target_link_libraries(ynotbit PRIVATE notbit_engine)
endif()
set_target_properties(ynotbit PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "ynotbit" MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.yshurik.ynotbit" MACOSX_BUNDLE_BUNDLE_VERSION "0.3.0")
install(TARGETS ynotbit BUNDLE DESTINATION . RUNTIME DESTINATION bin)
add_executable(lifecycle_tests tests/lifecycle_tests.cpp)
target_link_libraries(lifecycle_tests PRIVATE bmcore)
add_test(NAME lifecycle COMMAND lifecycle_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME relay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relay_test.py $<TARGET_FILE:ynotbit>)
 add_test(NAME relay-verack-first COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relay_test.py $<TARGET_FILE:ynotbit> verack-first)
 set_tests_properties(relay relay-verack-first PROPERTIES TIMEOUT 60)
 add_test(NAME qt-relay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relay_test.py $<TARGET_FILE:ynotbit>)
 add_test(NAME qt-relay-verack-first COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/relay_test.py $<TARGET_FILE:ynotbit> verack-first)
 set_tests_properties(qt-relay qt-relay-verack-first PROPERTIES ENVIRONMENT "YNOTBIT_TEST_NODE_FLAG=--qt-node" TIMEOUT 60)
endif()
add_test(NAME gui-smoke COMMAND ynotbit --offline --data-dir ${CMAKE_CURRENT_BINARY_DIR}/smoke-node --smoke-test)
set_tests_properties(gui-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
qt_add_executable(desktop_tests tests/desktop_tests.cpp src/session.cpp src/session.h)
qt_add_resources(desktop_tests "test-ui" PREFIX "/" FILES ui/Main.qml)
target_sources(desktop_tests PRIVATE src/appearance.cpp src/appearance.h src/markdown_editor.cpp src/markdown_editor.h)
target_link_libraries(desktop_tests PRIVATE bmcore Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME desktop COMMAND desktop_tests)
set_tests_properties(desktop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_executable(outbox_tests tests/outbox_tests.cpp)
target_link_libraries(outbox_tests PRIVATE bmcore)
add_test(NAME outbox COMMAND outbox_tests)

add_executable(wire_tests tests/wire_tests.cpp)
target_link_libraries(wire_tests PRIVATE bmcore)
add_test(NAME wire COMMAND wire_tests)

add_executable(delivery_tests tests/delivery_tests.cpp)
target_link_libraries(delivery_tests PRIVATE bmcore)
add_test(NAME delivery COMMAND delivery_tests)
set_tests_properties(delivery PROPERTIES TIMEOUT 120)

qt_add_executable(memory_tests tests/memory_tests.cpp src/session.cpp src/session.h)
target_link_libraries(memory_tests PRIVATE bmcore Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME memory COMMAND memory_tests)
set_tests_properties(memory PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
qt_add_executable(editor_tests tests/editor_tests.cpp src/appearance.cpp src/appearance.h src/markdown_editor.cpp src/markdown_editor.h)
target_link_libraries(editor_tests PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Test)
add_test(NAME editor COMMAND editor_tests)
set_tests_properties(editor PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
